Option Explicit

Private Const KOKPIT_SILINECEK_UST_SATIR_SAYISI As Long = 2

Public Sub Run_ButceKullanimOzeti()

    ButceKullanimOzeti_Core

End Sub


Public Sub Run_ButceKullanimOzeti_FromAll(ByVal pTedarikDosyaYolu As String, _
                                           ByVal pYatirimButceDosyaYolu As String, _
                                           ByVal pProjeBilgiDosyaYolu As String, _
                                           Optional ByVal pCiktiAnaKlasor As String = "", _
                                           Optional ByVal pButceTalepleriDosyaYolu As String = "")

    ButceKullanimOzeti_Core pTedarikDosyaYolu, _
                            pYatirimButceDosyaYolu, _
                            pProjeBilgiDosyaYolu, _
                            pCiktiAnaKlasor, _
                            pButceTalepleriDosyaYolu

End Sub


Private Sub ButceKullanimOzeti_Core(Optional ByVal pTedarikDosyaYolu As String = "", _
                                    Optional ByVal pYatirimButceDosyaYolu As String = "", _
                                    Optional ByVal pProjeBilgiDosyaYolu As String = "", _
                                    Optional ByVal pCiktiAnaKlasor As String = "", _
                                    Optional ByVal pButceTalepleriDosyaYolu As String = "")


   Dim wbTedarik As Workbook
   Dim wbYatirim As Workbook
   Dim wbYeni As Workbook

   Dim wsTedarik As Worksheet
   Dim wsYatirim As Worksheet
   Dim wsVeriSeti As Worksheet
   Dim wsKullanimOzet As Worksheet

   Dim tedarikDosyaYolu As String
   Dim yatirimDosyaYolu As String
   Dim projeBilgiDosyaYolu As String
   Dim butceTalepleriDosyaYolu As String

   Dim tarihMetni As String
   Dim ciktiKlasoru As String
   Dim ciktiDosyaYolu As String

   Dim tedarikHeaderRow As Long
   Dim yeniHeaderRow As Long

   Dim colTedButceNo As Long
   Dim colTedStatu As Long
   Dim colTedTutarUSD As Long

   Dim colYeniButceNo As Long
   Dim colYeniButce As Long
   Dim colYeniKullanilan As Long
   Dim colYeniBakiye As Long

   Dim colHesapHarcama As Long
   Dim colHesapTalep As Long

   Dim sonSatirTed As Long
   Dim sonSatirYatirim As Long
   Dim sonSutunYatirim As Long
   Dim sonSatirYeni As Long
   Dim sonSutunYeni As Long

   Dim dictHarcama As Object
   Dim dictTalep As Object

   Dim i As Long
   Dim butceNo As String
   Dim statu As String
   Dim tutarUSD As Double

   Dim eskiEkran As Boolean
   Dim eskiUyari As Boolean
   Dim hataMesaji As String

   On Error GoTo HataYonetimi

   eskiEkran = Application.ScreenUpdating
   eskiUyari = Application.DisplayAlerts

   Application.ScreenUpdating = False
   Application.DisplayAlerts = False

   tarihMetni = Format(Date, "dd.mm.yyyy")

   If Len(Trim(pTedarikDosyaYolu)) > 0 Then
       tedarikDosyaYolu = pTedarikDosyaYolu
   Else
       tedarikDosyaYolu = DosyaSec("Lütfen KOKPİT sisteminden alınan Tedarik Kalemleri dosyasını seçiniz.")
   End If

   If Len(Trim(tedarikDosyaYolu)) = 0 Then GoTo TemizCikis

   If Len(Trim(pYatirimButceDosyaYolu)) > 0 Then
       yatirimDosyaYolu = pYatirimButceDosyaYolu
   Else
       yatirimDosyaYolu = DosyaSec("Lütfen Yatırım Bütçe dosyasını seçiniz.")
   End If
   
    If Len(Trim(pProjeBilgiDosyaYolu)) > 0 Then
    projeBilgiDosyaYolu = pProjeBilgiDosyaYolu
    Else
        projeBilgiDosyaYolu = DosyaSec("Lütfen Proje Kodu ve Proje Adı bilgilerini içeren kaynak dosyayı seçiniz.")
    End If
    
    If Len(Trim(projeBilgiDosyaYolu)) = 0 Then GoTo TemizCikis

   If Len(Trim(pButceTalepleriDosyaYolu)) > 0 Then
       butceTalepleriDosyaYolu = pButceTalepleriDosyaYolu
   Else
       butceTalepleriDosyaYolu = DosyaSec("Lütfen Bütçe Talepleri dosyasını seçiniz.")
   End If

   If Len(Trim(butceTalepleriDosyaYolu)) = 0 Then GoTo TemizCikis

   If Len(Trim(yatirimDosyaYolu)) = 0 Then GoTo TemizCikis

   If Len(Trim(pCiktiAnaKlasor)) > 0 Then
       ciktiKlasoru = EnsureBudgetUsageFolder(pCiktiAnaKlasor, tarihMetni)
   Else
       ciktiKlasoru = EnsureBudgetUsageFolder(GetParentFolder(tedarikDosyaYolu), tarihMetni)
   End If

   Set wbTedarik = Workbooks.Open(tedarikDosyaYolu, ReadOnly:=True)
   Set wbYatirim = Workbooks.Open(yatirimDosyaYolu, ReadOnly:=True)

   Set wsTedarik = wbTedarik.Worksheets(1)
   Set wsYatirim = wbYatirim.Worksheets(1)

   ' ------------------------------------------------------------
   ' Tedarik dosyasından eski T/U algoritması için sözlükleri hazırla
   ' ------------------------------------------------------------
   tedarikHeaderRow = FindHeaderRow(wsTedarik, Array("Bütçe No", "Butce No"))
   If tedarikHeaderRow = 0 Then Err.Raise vbObjectError + 1001, , "Tedarik Kalemleri dosyasında 'Bütçe No' başlığı bulunamadı."

   colTedButceNo = FindHeaderCol(wsTedarik, tedarikHeaderRow, Array("Bütçe No", "Butce No", "Bütçe Numarası", "Butce Numarasi"))
   colTedStatu = FindHeaderCol(wsTedarik, tedarikHeaderRow, Array("Statü", "Statu", "Durum", "Talep Statüsü", "Talep Statusu", "Tedarik Statüsü", "Tedarik Statusu"))
   colTedTutarUSD = FindHeaderCol(wsTedarik, tedarikHeaderRow, Array( _
                                   "Talep Toplam USD", _
                                   "Toplam Talep USD", _
                                   "Toplam Talep Tutarı USD", _
                                   "Toplam Talep Tutari USD", _
                                   "Talep Tutarı USD", _
                                   "Talep Tutari USD", _
                                   "ABD $", _
                                   "USD"))

   If colTedButceNo = 0 Then Err.Raise vbObjectError + 1002, , "Tedarik Kalemleri dosyasında 'Bütçe No' kolonu bulunamadı."
   If colTedStatu = 0 Then Err.Raise vbObjectError + 1003, , "Tedarik Kalemleri dosyasında 'Statü' kolonu bulunamadı."
   If colTedTutarUSD = 0 Then Err.Raise vbObjectError + 1004, , "Tedarik Kalemleri dosyasında 'Talep Toplam USD' kolonu bulunamadı."

   Set dictHarcama = CreateObject("Scripting.Dictionary")
   Set dictTalep = CreateObject("Scripting.Dictionary")

   sonSatirTed = LastUsedRow(wsTedarik)

   For i = tedarikHeaderRow + 1 To sonSatirTed

       butceNo = NormalizeButceNo(wsTedarik.Cells(i, colTedButceNo).Value)
       statu = CStr(wsTedarik.Cells(i, colTedStatu).Value)
       tutarUSD = ToDoubleSafe(wsTedarik.Cells(i, colTedTutarUSD).Value)

       If Len(butceNo) > 0 Then

           If IsHarcamaStatus(statu) Then
               AddToDict dictHarcama, butceNo, tutarUSD
               AddToDict dictTalep, butceNo, tutarUSD

           ElseIf IsSadeceTalepStatus(statu) Then
               AddToDict dictTalep, butceNo, tutarUSD
           End If

       End If

   Next i

   ' ------------------------------------------------------------
   ' 01 - Yatırım Bütçe Kullanım Durum Özeti dosyası
   ' ------------------------------------------------------------
   ciktiDosyaYolu = ciktiKlasoru & Application.PathSeparator & _
                    "01 - Yatırım Bütçe Kullanım Durum Özeti - " & tarihMetni & ".xlsx"

   Set wbYeni = Workbooks.Add(xlWBATWorksheet)

   Set wsVeriSeti = wbYeni.Worksheets(1)
   wsVeriSeti.Name = "01 - Veri Seti"

   Set wsKullanimOzet = wbYeni.Worksheets.Add(After:=wsVeriSeti)
   wsKullanimOzet.Name = "02 - Kullanım Özeti"

   PrepareKokpitVeriSeti wsTedarik, wsVeriSeti, "YYSHUMA"

   sonSatirYatirim = LastUsedRow(wsYatirim)
   sonSutunYatirim = LastUsedCol(wsYatirim)

   wsYatirim.Range(wsYatirim.Cells(1, 1), wsYatirim.Cells(sonSatirYatirim, sonSutunYatirim)).Copy
   wsKullanimOzet.Range("A1").PasteSpecial xlPasteAll
   Application.CutCopyMode = False

   yeniHeaderRow = FindHeaderRow(wsKullanimOzet, Array("Bütçe No", "Butce No"))
   If yeniHeaderRow = 0 Then Err.Raise vbObjectError + 1005, , "Yatırım Bütçe dosyasında 'Bütçe No' başlığı bulunamadı."

   If yeniHeaderRow = 1 Then
       wsKullanimOzet.rows(1).Insert Shift:=xlDown
       yeniHeaderRow = yeniHeaderRow + 1
   End If

   DeleteColumnsByExactHeaders wsKullanimOzet, yeniHeaderRow, Array( _
       "Bütçe (TL)", _
       "Kullanılan(TL)", _
       "Kullanılan (TL)", _
       "Bakiye(TL)", _
       "Bakiye (TL)", _
       "%(TL)", _
       "% (TL)", _
       "Nakit TL", _
       "Üst Bütçe", _
       "Ust Butce")

   InsertMudurlukColumn wsKullanimOzet, yeniHeaderRow

   colYeniButceNo = FindHeaderCol(wsKullanimOzet, yeniHeaderRow, Array("Bütçe No", "Butce No", "Bütçe Numarası", "Butce Numarasi"))
   colYeniButce = FindHeaderCol(wsKullanimOzet, yeniHeaderRow, Array("Bütçe ($)", "Butce ($)", "Bütçe$", "Butce$", "Bütçe", "Butce"))
   colYeniKullanilan = FindHeaderCol(wsKullanimOzet, yeniHeaderRow, Array("Kullanılan($)", "Kullanilan($)", "Kullanılan ($)", "Kullanilan ($)", "Kullanılan", "Kullanilan"))
   colYeniBakiye = FindHeaderCol(wsKullanimOzet, yeniHeaderRow, Array("Bakiye($)", "Bakiye ($)", "Bakiye"))

   If colYeniButceNo = 0 Then Err.Raise vbObjectError + 1006, , "Yatırım Bütçe dosyasında 'Bütçe No' kolonu bulunamadı."
   If colYeniButce = 0 Then Err.Raise vbObjectError + 1007, , "Yatırım Bütçe dosyasında 'Bütçe ($)' kolonu bulunamadı."
   If colYeniKullanilan = 0 Then Err.Raise vbObjectError + 1008, , "Yatırım Bütçe dosyasında 'Kullanılan($)' kolonu bulunamadı."
   If colYeniBakiye = 0 Then Err.Raise vbObjectError + 1009, , "Yatırım Bütçe dosyasında 'Bakiye($)' kolonu bulunamadı."

   sonSutunYeni = LastUsedCol(wsKullanimOzet)

   colHesapHarcama = sonSutunYeni + 1
   colHesapTalep = sonSutunYeni + 2

   wsKullanimOzet.Cells(yeniHeaderRow, colHesapHarcama).Value = "Hesaplanan Harcama Tutarı USD"
   wsKullanimOzet.Cells(yeniHeaderRow, colHesapTalep).Value = "Hesaplanan Talep Toplam Tutarı USD"
   wsKullanimOzet.Cells(yeniHeaderRow, colHesapHarcama).Font.Bold = True
   wsKullanimOzet.Cells(yeniHeaderRow, colHesapTalep).Font.Bold = True

   sonSatirYeni = LastUsedRow(wsKullanimOzet)

   For i = yeniHeaderRow + 1 To sonSatirYeni

       butceNo = NormalizeButceNo(wsKullanimOzet.Cells(i, colYeniButceNo).Value)

       If Len(butceNo) > 0 Then

           If dictHarcama.Exists(butceNo) Then
               wsKullanimOzet.Cells(i, colHesapHarcama).Value = dictHarcama(butceNo)
           Else
               wsKullanimOzet.Cells(i, colHesapHarcama).Value = 0
           End If

           If dictTalep.Exists(butceNo) Then
               wsKullanimOzet.Cells(i, colHesapTalep).Value = dictTalep(butceNo)
           Else
               wsKullanimOzet.Cells(i, colHesapTalep).Value = 0
           End If

       End If

   Next i

   wsKullanimOzet.Cells(yeniHeaderRow - 1, colYeniButce).Formula = "=SUM(" & _
       wsKullanimOzet.Range(wsKullanimOzet.Cells(yeniHeaderRow + 1, colYeniButce), wsKullanimOzet.Cells(sonSatirYeni, colYeniButce)).Address(False, False) & ")"

   wsKullanimOzet.Cells(yeniHeaderRow - 1, colYeniKullanilan).Formula = "=SUM(" & _
       wsKullanimOzet.Range(wsKullanimOzet.Cells(yeniHeaderRow + 1, colYeniKullanilan), wsKullanimOzet.Cells(sonSatirYeni, colYeniKullanilan)).Address(False, False) & ")"

   wsKullanimOzet.Cells(yeniHeaderRow - 1, colYeniBakiye).Formula = "=SUM(" & _
       wsKullanimOzet.Range(wsKullanimOzet.Cells(yeniHeaderRow + 1, colYeniBakiye), wsKullanimOzet.Cells(sonSatirYeni, colYeniBakiye)).Address(False, False) & ")"

   wsKullanimOzet.Cells(yeniHeaderRow - 1, colHesapHarcama).Formula = "=SUM(" & _
       wsKullanimOzet.Range(wsKullanimOzet.Cells(yeniHeaderRow + 1, colHesapHarcama), wsKullanimOzet.Cells(sonSatirYeni, colHesapHarcama)).Address(False, False) & ")"

   wsKullanimOzet.Cells(yeniHeaderRow - 1, colHesapTalep).Formula = "=SUM(" & _
       wsKullanimOzet.Range(wsKullanimOzet.Cells(yeniHeaderRow + 1, colHesapTalep), wsKullanimOzet.Cells(sonSatirYeni, colHesapTalep)).Address(False, False) & ")"

   wsKullanimOzet.Range(wsKullanimOzet.Cells(yeniHeaderRow - 1, colYeniButce), _
                        wsKullanimOzet.Cells(yeniHeaderRow - 1, colHesapTalep)).Font.Bold = True

   ApplyNumberFormat wsKullanimOzet, yeniHeaderRow, sonSatirYeni, _
       Array(colYeniButce, colYeniKullanilan, colYeniBakiye, colHesapHarcama, colHesapTalep)

   wsKullanimOzet.Range(wsKullanimOzet.Cells(yeniHeaderRow, 1), wsKullanimOzet.Cells(yeniHeaderRow, colHesapTalep)).Font.Bold = True
   wsKullanimOzet.Range(wsKullanimOzet.Cells(yeniHeaderRow, 1), wsKullanimOzet.Cells(sonSatirYeni, colHesapTalep)).Borders.LineStyle = xlContinuous

   If wsKullanimOzet.AutoFilterMode Then wsKullanimOzet.AutoFilterMode = False
   wsKullanimOzet.Range(wsKullanimOzet.Cells(yeniHeaderRow, 1), wsKullanimOzet.Cells(sonSatirYeni, colHesapTalep)).AutoFilter

   wsKullanimOzet.Columns.AutoFit

    AutoFitAllSheets wbYeni
   wbYeni.SaveAs Filename:=ciktiDosyaYolu, FileFormat:=xlOpenXMLWorkbook
   wbYeni.Close SaveChanges:=False
   Set wbYeni = Nothing

   ' ------------------------------------------------------------
   ' 02 - İşletme Bütçe Kullanım Durum Özeti dosyası
   ' ------------------------------------------------------------
   CreateIsletmeButceKullanimDosyasi wsTedarik, ciktiKlasoru, tarihMetni, projeBilgiDosyaYolu, butceTalepleriDosyaYolu

   ' ------------------------------------------------------------
   ' 03 - Gider Adı Tasnifli İşletme dosyaları
   ' ------------------------------------------------------------
   CreateGiderAdiTasnifliIsletmeDosyalari wsTedarik, ciktiKlasoru, tarihMetni, butceTalepleriDosyaYolu, projeBilgiDosyaYolu

TemizCikis:
   On Error Resume Next

   If Not wbYeni Is Nothing Then wbYeni.Close SaveChanges:=False
   If Not wbTedarik Is Nothing Then wbTedarik.Close SaveChanges:=False
   If Not wbYatirim Is Nothing Then wbYatirim.Close SaveChanges:=False

   Application.DisplayAlerts = eskiUyari
   Application.ScreenUpdating = eskiEkran

   If Len(Trim(ciktiKlasoru)) > 0 Then
       MsgBox "Bütçe Kullanım Durumu dosyaları başarıyla oluşturuldu." & vbCrLf & vbCrLf & _
              "Kayıt klasörü:" & vbCrLf & ciktiKlasoru, vbInformation, "İşlem Tamamlandı"
   End If

   Exit Sub

HataYonetimi:
   hataMesaji = Err.Description
   If Len(Trim(hataMesaji)) = 0 Then hataMesaji = "Bilinmeyen hata. Lütfen kaynak dosya başlıklarını kontrol edin."
   hataMesaji = "Hata No: " & Err.Number & vbCrLf & hataMesaji

   On Error Resume Next

   If Not wbYeni Is Nothing Then wbYeni.Close SaveChanges:=False
   If Not wbTedarik Is Nothing Then wbTedarik.Close SaveChanges:=False
   If Not wbYatirim Is Nothing Then wbYatirim.Close SaveChanges:=False

   Application.DisplayAlerts = eskiUyari
   Application.ScreenUpdating = eskiEkran

   MsgBox "İşlem sırasında hata oluştu:" & vbCrLf & vbCrLf & _
          hataMesaji, vbCritical, "Bütçe Kullanım Özeti"

End Sub

Private Sub CreateIsletmeButceKullanimDosyasi(ByVal wsTedarik As Worksheet, _
                                               ByVal ciktiKlasoru As String, _
                                               ByVal tarihMetni As String, _
                                               ByVal projeBilgiDosyaYolu As String, _
                                               ByVal butceTalepleriDosyaYolu As String)


   Dim wbIsletme As Workbook
   Dim wsVeriSeti As Worksheet
   Dim wsGerceklesen As Worksheet
   Dim wsTalep As Worksheet
   Dim wsProjeGerceklesen As Worksheet
   Dim wsProjeTalep As Worksheet
   Dim wsMudurlukGerceklesen As Worksheet
   Dim wsMudurlukTalep As Worksheet
   Dim ciktiDosyaYolu As String
   Dim wbProjeBilgi As Workbook
   Dim wsProjeBilgi As Worksheet
   Dim wbButceTalepleri As Workbook
   Dim wsButceTipleri As Worksheet
   Dim wsButceKaynak As Worksheet
   Dim wsButceTipleriKopya As Worksheet
   Dim dictProjeAdi As Object
   Dim dictKullanilacakButceTipleri As Object
   Dim dictProjeButce As Object
   Dim dictMudurlukProjeButce As Object
   Dim dictProjectFinName As Object
   Dim dictProjeGiderAdi As Object


   ciktiDosyaYolu = ciktiKlasoru & Application.PathSeparator & _
                    "02 - İşletme Bütçe Kullanım Durum Özeti - " & tarihMetni & ".xlsx"
    Set wbProjeBilgi = Workbooks.Open(projeBilgiDosyaYolu, ReadOnly:=True)
    Set wsProjeBilgi = wbProjeBilgi.Worksheets(1)
    Set dictProjeAdi = BuildProjeAdiDict(wsProjeBilgi)
    Set wbButceTalepleri = Workbooks.Open(butceTalepleriDosyaYolu, ReadOnly:=True)
    Set wsButceTipleri = FindWorksheetByName(wbButceTalepleri, "Kullanılacak Bütçe Tipleri")

    If wsButceTipleri Is Nothing Then
        Err.Raise vbObjectError + 1325, , "Bütçe Talepleri dosyasında 'Kullanılacak Bütçe Tipleri' çalışma sayfası bulunamadı."
    End If

    Set wsButceKaynak = FindWorksheetWithHeaders(wbButceTalepleri, Array("Proje", "Gider Adı", "Grup", "Toplam USD"))
    If wsButceKaynak Is Nothing Then
        Err.Raise vbObjectError + 1326, , "Bütçe Talepleri dosyasında 'Proje', 'Gider Adı', 'Grup' ve 'Toplam USD' sütunlarını içeren kaynak sayfa bulunamadı."
    End If

    Set dictKullanilacakButceTipleri = BuildKullanilacakButceTipleriDict(wsButceTipleri)
    Set dictProjeButce = BuildProjeButceDict(wsButceKaynak, dictKullanilacakButceTipleri, dictProjeAdi)
    Set dictMudurlukProjeButce = BuildMudurlukProjeButceDict(wsButceKaynak, dictKullanilacakButceTipleri)
    Set dictProjectFinName = BuildProjectFinNameDict(wsProjeBilgi)
    Set dictProjeGiderAdi = BuildButceTalepleriProjeGiderDict(wsButceKaynak)

   Set wbIsletme = Workbooks.Add(xlWBATWorksheet)

   Set wsVeriSeti = wbIsletme.Worksheets(1)
   wsVeriSeti.Name = "01 - Veri Seti"

   wsButceTipleri.Copy After:=wsVeriSeti
   Set wsButceTipleriKopya = wbIsletme.Worksheets(wsVeriSeti.Index + 1)
   wsButceTipleriKopya.Name = "02-Kullanılacak Bütçe Tipleri"

   Set wsProjeGerceklesen = wbIsletme.Worksheets.Add(After:=wsButceTipleriKopya)
   wsProjeGerceklesen.Name = "03- V4000-GERÇEKLEŞEN"

   Set wsProjeTalep = wbIsletme.Worksheets.Add(After:=wsProjeGerceklesen)
   wsProjeTalep.Name = "04- V4000 - TALEP"

   Set wsGerceklesen = wbIsletme.Worksheets.Add(After:=wsProjeTalep)
   wsGerceklesen.Name = "05- BİRİMLER-GERÇEKLEŞEN"

   Set wsTalep = wbIsletme.Worksheets.Add(After:=wsGerceklesen)
   wsTalep.Name = "06-BİRİMLER-TALEP"

   Set wsMudurlukGerceklesen = wbIsletme.Worksheets.Add(After:=wsTalep)
   wsMudurlukGerceklesen.Name = "07-MÜDÜRLÜK-GERÇEKLEŞEN"

   Set wsMudurlukTalep = wbIsletme.Worksheets.Add(After:=wsMudurlukGerceklesen)
   wsMudurlukTalep.Name = "08-MÜDÜRLÜK TALEP"

   PrepareKokpitVeriSeti wsTedarik, wsVeriSeti, "IPTSHUMA"

    CreateIsletmeKullanimOzeti wsVeriSeti, wsProjeGerceklesen, False, "Gerçekleşen", dictProjeAdi, True, dictProjeButce, dictProjectFinName, dictProjeGiderAdi
    CreateIsletmeKullanimOzeti wsVeriSeti, wsProjeTalep, True, "Talep", dictProjeAdi, True, dictProjeButce, dictProjectFinName, dictProjeGiderAdi
    CreateIsletmeKullanimOzeti wsVeriSeti, wsGerceklesen, False, "Gerçekleşen", dictProjeAdi, False, Nothing, dictProjectFinName, dictProjeGiderAdi
    CreateIsletmeKullanimOzeti wsVeriSeti, wsTalep, True, "Talep", dictProjeAdi, False, Nothing, dictProjectFinName, dictProjeGiderAdi
    CreateIsletmeKullanimOzeti wsVeriSeti, wsMudurlukGerceklesen, False, "Gerçekleşen", dictProjeAdi, False, dictProjeButce, dictProjectFinName, dictProjeGiderAdi, True, dictMudurlukProjeButce
    CreateIsletmeKullanimOzeti wsVeriSeti, wsMudurlukTalep, True, "Talep", dictProjeAdi, False, dictProjeButce, dictProjectFinName, dictProjeGiderAdi, True, dictMudurlukProjeButce


   AutoFitAllSheets wbIsletme
   wbIsletme.SaveAs Filename:=ciktiDosyaYolu, FileFormat:=xlOpenXMLWorkbook
   wbIsletme.Close SaveChanges:=False
   If Not wbProjeBilgi Is Nothing Then wbProjeBilgi.Close SaveChanges:=False
   If Not wbButceTalepleri Is Nothing Then wbButceTalepleri.Close SaveChanges:=False

End Sub


Private Sub CreateIsletmeKullanimOzeti(ByVal wsVeriSeti As Worksheet, _
                                      ByVal wsOzet As Worksheet, _
                                      ByVal humaStatuleriDahil As Boolean, _
                                      ByVal baslikTipi As String, _
                                      ByVal dictProjeAdi As Object, _
                                      Optional ByVal projeBazliOzet As Boolean = False, _
                                      Optional ByVal dictProjeButce As Object = Nothing, _
                                      Optional ByVal dictProjectFinName As Object = Nothing, _
                                      Optional ByVal dictProjeGiderAdi As Object = Nothing, _
                                      Optional ByVal mudurlukBazliOzet As Boolean = False, _
                                      Optional ByVal dictMudurlukProjeButce As Object = Nothing)

   Dim headerRow As Long
   Dim lastRow As Long
   Dim i As Long

   Dim colTalepEdenBolum As Long
   Dim colProje As Long
   Dim colStatu As Long
   Dim colGirisTarihi As Long
   Dim colTalepToplamUSD As Long

   Dim dict As Object
   Dim key As String

   Dim talepEdenBolum As String
   Dim temizBirim As String
   Dim baskanlik As String
   Dim mudurluk As String
   Dim birim As String
   Dim proje As String
   Dim projeAdi As String
   Dim projectFinName As String
   Dim giderAdlari As String
   Dim statu As String
   Dim ayNo As Integer
   Dim tutar As Double
   Dim toplam As Double

   Dim arr As Variant
   Dim satir As Long
   Dim k As Variant
   Dim m As Long
   Dim lastOzetRow As Long

   Set dict = CreateObject("Scripting.Dictionary")

   headerRow = 1
   lastRow = LastUsedRow(wsVeriSeti)

   colTalepEdenBolum = FindHeaderCol(wsVeriSeti, headerRow, Array("Talep Eden Bölüm", "Talep Eden Bolum"))
   colProje = FindHeaderCol(wsVeriSeti, headerRow, Array("Proje"))
   colStatu = FindHeaderCol(wsVeriSeti, headerRow, Array("Statü", "Statu", "Durum", "Talep Statüsü", "Talep Statusu", "Tedarik Statüsü", "Tedarik Statusu"))
   colGirisTarihi = FindHeaderCol(wsVeriSeti, headerRow, Array("Giriş Tarihi", "Giris Tarihi"))
   colTalepToplamUSD = FindHeaderCol(wsVeriSeti, headerRow, Array("Talep Toplam USD"))

   If colTalepEdenBolum = 0 Then Err.Raise vbObjectError + 1101, , "'Talep Eden Bölüm' sütunu bulunamadı."
   If colProje = 0 Then Err.Raise vbObjectError + 1102, , "'Proje' sütunu bulunamadı."
   If colStatu = 0 Then Err.Raise vbObjectError + 1103, , "'Statü' sütunu bulunamadı."
   If colGirisTarihi = 0 Then Err.Raise vbObjectError + 1104, , "'Giriş Tarihi' sütunu bulunamadı."
   If colTalepToplamUSD = 0 Then Err.Raise vbObjectError + 1105, , "'Talep Toplam USD' sütunu bulunamadı."

   For i = headerRow + 1 To lastRow

       statu = CStr(wsVeriSeti.Cells(i, colStatu).Value)

       If IsIsletmeOzetStatus(statu, humaStatuleriDahil) Then

           ayNo = GetMonthNoFromValue(wsVeriSeti.Cells(i, colGirisTarihi).Value)

           If ayNo >= 1 And ayNo <= 12 Then

               talepEdenBolum = CStr(wsVeriSeti.Cells(i, colTalepEdenBolum).Value)
               temizBirim = CleanTalepEdenBolum(talepEdenBolum)

               If projeBazliOzet Then
                   baskanlik = "V"
                   mudurluk = "V4000"
                   birim = ""
               Else
                   If Len(temizBirim) >= 1 Then
                       baskanlik = Left(temizBirim, 1)
                   Else
                       baskanlik = ""
                   End If

                   If mudurlukBazliOzet Then
                       mudurluk = NormalizeMudurlukKodu(temizBirim)
                       birim = ""
                   Else
                       If Len(temizBirim) >= 3 Then
                           mudurluk = Left(temizBirim, 3)
                       Else
                           mudurluk = temizBirim
                       End If

                       birim = temizBirim
                   End If
               End If

               proje = Trim(CStr(wsVeriSeti.Cells(i, colProje).Value))
               tutar = ToDoubleSafe(wsVeriSeti.Cells(i, colTalepToplamUSD).Value)

               projeAdi = GetProjeAdiFromDict(dictProjeAdi, proje)

               If Not dictProjectFinName Is Nothing Then
                   ' 03/04 V4000 ve 05/06 Birimler sayfalarında Gider Adı Tasnifli dosyalardaki proje algoritması kullanılır:
                   ' KOKPİT Proje kodu -> Proje Bilgileri PROJECT_CODE/PROJECT_FIN_NAME -> Bütçe Talepleri Proje eşleşmesi.
                   projectFinName = GetProjectFinNameFromDict(dictProjectFinName, proje)
                   giderAdlari = GetGiderAdlariFromProjectDict(dictProjeGiderAdi, projectFinName)

                   If Len(projectFinName) > 0 And projectFinName <> NormalizeProjectCode(proje) Then
                       projeAdi = projectFinName
                   End If
               End If

               If projeBazliOzet Then
                   ' Proje bazlı özetlerde aynı proje tek satırdır.
                   key = NormalizeProjectCode(proje)
               ElseIf mudurlukBazliOzet Then
                   ' Müdürlük bazlı özetlerde aynı Müdürlük + Proje Adı tek satırdır.
                   ' Böylece aynı proje adı aynı müdürlükte birden fazla KOKPİT satırında/PROJECT_CODE altında gelirse aylık tutarlar toplanır.
                   key = mudurluk & "|" & NormalizeProjectNameKey(projeAdi)
               Else
                   ' Aynı Birim + Proje tek satır.
                   ' Aynı proje farklı birimdeyse ayrı satır.
                   key = birim & "|" & proje
               End If

               If Not dict.Exists(key) Then

                   ReDim arr(1 To 17)

                   arr(1) = baskanlik
                   arr(2) = mudurluk
                   arr(3) = birim
                   arr(4) = proje
                   arr(5) = projeAdi

                   For m = 6 To 17
                       arr(m) = 0#
                   Next m

                   dict.Add key, arr

               End If

               arr = dict(key)

               ' arr(6) = Ocak
               ' arr(17) = Aralık
               arr(5 + ayNo) = CDbl(arr(5 + ayNo)) + tutar

               dict(key) = arr

           End If

       End If

   Next i

   wsOzet.Cells.Clear

   ' ------------------------------------------------------------
   ' Nihai başlık düzeni
   ' ------------------------------------------------------------
   wsOzet.Range("A2").Value = "Gider Adı"
   wsOzet.Range("B2").Value = "Grup"
   wsOzet.Range("C2").Value = "Müdürlük"
   wsOzet.Range("D2").Value = "Birim"
   wsOzet.Range("E2").Value = "Pool"
   wsOzet.Range("F2").Value = "Proje"
   wsOzet.Range("G2").Value = "Proje Adı"
   wsOzet.Range("H2").Value = "Açıklama"
   wsOzet.Range("I2").Value = "Başkanlık"
   wsOzet.Range("J2").Value = "Toplam USD Bütçe"
   wsOzet.Range("K2").Value = baslikTipi & " Toplam USD"
   wsOzet.Range("L2").Value = "Kalan Bütçe USD"

   wsOzet.Range("M2").Value = "Ocak " & baslikTipi & " USD"
   wsOzet.Range("N2").Value = "Şubat " & baslikTipi & " USD"
   wsOzet.Range("O2").Value = "Mart " & baslikTipi & " USD"
   wsOzet.Range("P2").Value = "Nisan " & baslikTipi & " USD"
   wsOzet.Range("Q2").Value = "Mayıs " & baslikTipi & " USD"
   wsOzet.Range("R2").Value = "Haziran " & baslikTipi & " USD"
   wsOzet.Range("S2").Value = "Temmuz " & baslikTipi & " USD"
   wsOzet.Range("T2").Value = "Ağustos " & baslikTipi & " USD"
   wsOzet.Range("U2").Value = "Eylül " & baslikTipi & " USD"
   wsOzet.Range("V2").Value = "Ekim " & baslikTipi & " USD"
   wsOzet.Range("W2").Value = "Kasım " & baslikTipi & " USD"
   wsOzet.Range("X2").Value = "Aralık " & baslikTipi & " USD"

   wsOzet.Range("A2:X2").Font.Bold = True

   satir = 3

   For Each k In dict.Keys

       arr = dict(k)

       wsOzet.Cells(satir, "A").Value = ""
       wsOzet.Cells(satir, "B").Value = ""
       wsOzet.Cells(satir, "C").Value = arr(2) ' Müdürlük
       wsOzet.Cells(satir, "D").Value = arr(3) ' Birim
       wsOzet.Cells(satir, "E").Value = ""
       wsOzet.Cells(satir, "F").Value = arr(4) ' Proje
       wsOzet.Cells(satir, "G").Value = arr(5) ' Proje Adı
       wsOzet.Cells(satir, "H").Value = ""
       wsOzet.Cells(satir, "I").Value = arr(1) ' Başkanlık
       If mudurlukBazliOzet And Not dictMudurlukProjeButce Is Nothing Then
           wsOzet.Cells(satir, "J").Value = GetMudurlukProjeButceFromDict(dictMudurlukProjeButce, arr(2), arr(5))
       ElseIf Not dictProjeButce Is Nothing Then
           wsOzet.Cells(satir, "J").Value = GetProjeButceFromDict(dictProjeButce, arr(4), arr(5))
       Else
           wsOzet.Cells(satir, "J").Value = ""
       End If

       toplam = 0#

       For m = 1 To 12

           ' M = Ocak, X = Aralık
           wsOzet.Cells(satir, 12 + m).Value = arr(5 + m)
           toplam = toplam + CDbl(arr(5 + m))

       Next m

       ' K = tüm ayların toplamı
       wsOzet.Cells(satir, "K").Value = toplam

       ' L = J - K
       wsOzet.Cells(satir, "L").Formula = "=J" & satir & "-K" & satir

       satir = satir + 1

   Next k

   lastOzetRow = satir - 1

   If lastOzetRow >= 3 Then

       ' İlk satır toplamları
       wsOzet.Range("J1").Formula = "=SUM(J3:J" & lastOzetRow & ")"
       wsOzet.Range("K1").Formula = "=SUM(K3:K" & lastOzetRow & ")"
       wsOzet.Range("L1").Formula = "=J1-K1"

       wsOzet.Range("M1").Formula = "=SUM(M3:M" & lastOzetRow & ")"
       wsOzet.Range("N1").Formula = "=SUM(N3:N" & lastOzetRow & ")"
       wsOzet.Range("O1").Formula = "=SUM(O3:O" & lastOzetRow & ")"
       wsOzet.Range("P1").Formula = "=SUM(P3:P" & lastOzetRow & ")"
       wsOzet.Range("Q1").Formula = "=SUM(Q3:Q" & lastOzetRow & ")"
       wsOzet.Range("R1").Formula = "=SUM(R3:R" & lastOzetRow & ")"
       wsOzet.Range("S1").Formula = "=SUM(S3:S" & lastOzetRow & ")"
       wsOzet.Range("T1").Formula = "=SUM(T3:T" & lastOzetRow & ")"
       wsOzet.Range("U1").Formula = "=SUM(U3:U" & lastOzetRow & ")"
       wsOzet.Range("V1").Formula = "=SUM(V3:V" & lastOzetRow & ")"
       wsOzet.Range("W1").Formula = "=SUM(W3:W" & lastOzetRow & ")"
       wsOzet.Range("X1").Formula = "=SUM(X3:X" & lastOzetRow & ")"

       wsOzet.Range("J1:X1").Font.Bold = True

       With wsOzet.Sort
           .SortFields.Clear

           ' Tüm özetlerde Birim sütunu A-Z sıralanır; ardından Müdürlük ve Proje gelir.
           .SortFields.Add key:=wsOzet.Range("D3:D" & lastOzetRow), Order:=xlAscending
           .SortFields.Add key:=wsOzet.Range("C3:C" & lastOzetRow), Order:=xlAscending
           .SortFields.Add key:=wsOzet.Range("F3:F" & lastOzetRow), Order:=xlAscending

           .SetRange wsOzet.Range("A2:X" & lastOzetRow)
           .Header = xlYes
           .Apply
       End With

   Else

       wsOzet.Range("J1").Value = 0
       wsOzet.Range("K1").Value = 0
       wsOzet.Range("L1").Formula = "=J1-K1"
       wsOzet.Range("M1:X1").Value = 0
       wsOzet.Range("J1:X1").Font.Bold = True

   End If

   ' Finansal format:
   ' J, K, L ve M-X finansal alanlar.
   wsOzet.Range("J1:X" & Application.Max(lastOzetRow, 3)).NumberFormat = "#,##0.00 ""USD"""

   wsOzet.Range("A2:X" & Application.Max(lastOzetRow, 2)).Borders.LineStyle = xlContinuous
   wsOzet.Columns("A:X").AutoFit

   If wsOzet.AutoFilterMode Then wsOzet.AutoFilterMode = False
   wsOzet.Range("A2:X" & Application.Max(lastOzetRow, 2)).AutoFilter

End Sub

Private Sub CreateGiderAdiTasnifliDosya(ByVal wsVeriSeti As Worksheet, _
                                        ByVal dictGiderAdi As Object, _
                                        ByVal dictProjeGiderAdi As Object, _
                                        ByVal dictProjectFinName As Object, _
                                        ByVal dosyaYolu As String, _
                                        ByVal humaStatuleriDahil As Boolean, _
                                        ByVal actualHeader As String, _
                                        ByVal totalActualHeader As String)

   Dim wbYeni As Workbook
   Dim wsBolum As Worksheet
   Dim dictBolumler As Object
   Dim dictTutarlar As Object
   Dim bolumKey As Variant
   Dim giderKey As Variant
   Dim sheetName As String
   Dim rowNo As Long
   Dim totalRow As Long
   Dim monthNo As Long
   Dim budgetCol As Long
   Dim actualCol As Long
   Dim tutarKey As String
   Dim budgetTotalFormula As String
   Dim actualTotalFormula As String
   Dim aylar As Variant
   Dim lastDataRow As Long

   Set dictBolumler = BuildTasnifliBolumDict(wsVeriSeti, humaStatuleriDahil)
   Set dictTutarlar = BuildTasnifliTutarDict(wsVeriSeti, humaStatuleriDahil, dictProjeGiderAdi, dictProjectFinName)

   Set wbYeni = Workbooks.Add(xlWBATWorksheet)

   If dictBolumler.Count = 0 Then
       wbYeni.Worksheets(1).Name = "Veri Yok"
   Else
       Do While wbYeni.Worksheets.Count < dictBolumler.Count
           wbYeni.Worksheets.Add After:=wbYeni.Worksheets(wbYeni.Worksheets.Count)
       Loop

       Do While wbYeni.Worksheets.Count > dictBolumler.Count
           wbYeni.Worksheets(wbYeni.Worksheets.Count).Delete
       Loop

       aylar = Array("Ocak", "Şubat", "Mart", "Nisan", "Mayıs", "Haziran", "Temmuz", "Ağustos", "Eylül", "Ekim", "Kasım", "Aralık")

       For Each bolumKey In dictBolumler.Keys
           Set wsBolum = wbYeni.Worksheets(CLng(dictBolumler(bolumKey)))
           sheetName = GetUniqueWorksheetName(wbYeni, CleanWorksheetName(CStr(bolumKey)), wsBolum)
           wsBolum.Name = sheetName
           wsBolum.Cells.Clear

           wsBolum.Range("A1").Value = CStr(bolumKey)
           wsBolum.Range("A2").Value = ""

           For monthNo = 1 To 12
               budgetCol = 2 + ((monthNo - 1) * 2)
               actualCol = budgetCol + 1
               wsBolum.Cells(1, budgetCol).Value = aylar(monthNo - 1)
               wsBolum.Cells(1, actualCol).Value = aylar(monthNo - 1)
               wsBolum.Cells(2, budgetCol).Value = "Bütçe"
               wsBolum.Cells(2, actualCol).Value = actualHeader
           Next monthNo

           With wsBolum.Range("Z1:AA1")
               .Merge
               .Value = "TOPLAM"
               .HorizontalAlignment = xlCenter
           End With
           wsBolum.Range("Z2").Value = "Toplam Bütçe"
           wsBolum.Range("AA2").Value = totalActualHeader

           rowNo = 3
           For Each giderKey In dictGiderAdi.Keys
               wsBolum.Cells(rowNo, 1).Value = CStr(giderKey)

               For monthNo = 1 To 12
                   actualCol = 3 + ((monthNo - 1) * 2)
                   tutarKey = NormalizeText(CStr(bolumKey)) & "|" & NormalizeText(CStr(giderKey)) & "|" & CStr(monthNo)

                   If dictTutarlar.Exists(tutarKey) Then
                       wsBolum.Cells(rowNo, actualCol).Value = dictTutarlar(tutarKey)
                   Else
                       wsBolum.Cells(rowNo, actualCol).Value = 0
                   End If
               Next monthNo

               budgetTotalFormula = "=SUM(B" & rowNo & ",D" & rowNo & ",F" & rowNo & ",H" & rowNo & ",J" & rowNo & ",L" & rowNo & ",N" & rowNo & ",P" & rowNo & ",R" & rowNo & ",T" & rowNo & ",V" & rowNo & ",X" & rowNo & ")"
               actualTotalFormula = "=SUM(C" & rowNo & ",E" & rowNo & ",G" & rowNo & ",I" & rowNo & ",K" & rowNo & ",M" & rowNo & ",O" & rowNo & ",Q" & rowNo & ",S" & rowNo & ",U" & rowNo & ",W" & rowNo & ",Y" & rowNo & ")"
               wsBolum.Cells(rowNo, 26).Formula = budgetTotalFormula
               wsBolum.Cells(rowNo, 27).Formula = actualTotalFormula

               rowNo = rowNo + 1
           Next giderKey

           lastDataRow = rowNo - 1
           totalRow = rowNo

           ApplyUsdAccountingFormat wsBolum, 3, lastDataRow, 2, 27

           For monthNo = 1 To 12
               budgetCol = 2 + ((monthNo - 1) * 2)
               actualCol = budgetCol + 1

               With wsBolum.Range(wsBolum.Cells(totalRow, budgetCol), wsBolum.Cells(totalRow, actualCol))
                   .Merge
                   If lastDataRow >= 3 Then
                       .Cells(1, 1).Formula = "=IFERROR(SUM(" & wsBolum.Range(wsBolum.Cells(3, actualCol), wsBolum.Cells(lastDataRow, actualCol)).Address(False, False) & ")/SUM(" & wsBolum.Range(wsBolum.Cells(3, budgetCol), wsBolum.Cells(lastDataRow, budgetCol)).Address(False, False) & "),0)"
                   Else
                       .Cells(1, 1).Value = ""
                   End If
                   .NumberFormat = "0.00%"
                   .Font.Bold = True
                   .HorizontalAlignment = xlCenter
               End With
           Next monthNo

           With wsBolum.Range(wsBolum.Cells(totalRow, 26), wsBolum.Cells(totalRow, 27))
               .Merge
               If lastDataRow >= 3 Then
                   .Cells(1, 1).Formula = "=IFERROR(SUM(" & wsBolum.Range(wsBolum.Cells(3, 27), wsBolum.Cells(lastDataRow, 27)).Address(False, False) & ")/SUM(" & wsBolum.Range(wsBolum.Cells(3, 26), wsBolum.Cells(lastDataRow, 26)).Address(False, False) & "),0)"
               Else
                   .Cells(1, 1).Value = ""
               End If
               .NumberFormat = "0.00%"
               .Font.Bold = True
               .HorizontalAlignment = xlCenter
           End With

           wsBolum.Range(wsBolum.Cells(1, 1), wsBolum.Cells(2, 27)).Font.Bold = True
           wsBolum.Range(wsBolum.Cells(1, 1), wsBolum.Cells(totalRow, 27)).Borders.LineStyle = xlContinuous
           wsBolum.Columns.AutoFit
       Next bolumKey
   End If

   SortWorksheetsAZ wbYeni

   wbYeni.SaveAs Filename:=dosyaYolu, FileFormat:=xlOpenXMLWorkbook
   wbYeni.Close SaveChanges:=False

End Sub

Private Function BuildTasnifliBolumDict(ByVal wsVeriSeti As Worksheet, ByVal humaStatuleriDahil As Boolean) As Object

   Dim dict As Object
   Dim headerRow As Long
   Dim lastRow As Long
   Dim colBolum As Long
   Dim colStatu As Long
   Dim i As Long
   Dim bolum As String
   Dim statu As String

   Set dict = CreateObject("Scripting.Dictionary")
   dict.Add "0V4", 1

   headerRow = 1
   colBolum = FindHeaderCol(wsVeriSeti, headerRow, Array("Talep Eden Bölüm", "Talep Eden Bolum"))
   colStatu = FindHeaderCol(wsVeriSeti, headerRow, Array("Statü", "Statu", "Durum", "Talep Statüsü", "Talep Statusu", "Tedarik Statüsü", "Tedarik Statusu"))

   If colBolum = 0 Then Err.Raise vbObjectError + 1402, , "KOKPİT veri setinde 'Talep Eden Bölüm' sütunu bulunamadı."
   If colStatu = 0 Then Err.Raise vbObjectError + 1403, , "KOKPİT veri setinde 'Statü' sütunu bulunamadı."

   lastRow = LastUsedRow(wsVeriSeti)

   For i = headerRow + 1 To lastRow
       statu = CStr(wsVeriSeti.Cells(i, colStatu).Value)

       If IsIsletmeOzetStatus(statu, humaStatuleriDahil) Then
           bolum = Trim(CStr(wsVeriSeti.Cells(i, colBolum).Value))
           If Len(bolum) > 0 Then
               If Not dict.Exists(bolum) Then dict.Add bolum, dict.Count + 1
           End If
       End If
   Next i

   Set BuildTasnifliBolumDict = dict

End Function

Private Function BuildTasnifliTutarDict(ByVal wsVeriSeti As Worksheet, _
                                      ByVal humaStatuleriDahil As Boolean, _
                                      ByVal dictProjeGiderAdi As Object, _
                                      ByVal dictProjectFinName As Object) As Object

   Dim dict As Object
   Dim headerRow As Long
   Dim lastRow As Long
   Dim colBolum As Long
   Dim colProje As Long
   Dim colGiderAdi As Long
   Dim colStatu As Long
   Dim colGirisTarihi As Long
   Dim colTalepToplamUSD As Long
   Dim i As Long
   Dim bolum As String
   Dim projeKodu As String
   Dim projectFinName As String
   Dim giderAdi As String
   Dim giderAdlari As String
   Dim giderParcalari As Variant
   Dim giderItem As Variant
   Dim statu As String
   Dim ayNo As Integer
   Dim tutar As Double
   Dim key As String

   Set dict = CreateObject("Scripting.Dictionary")

   headerRow = 1
   colBolum = FindHeaderCol(wsVeriSeti, headerRow, Array("Talep Eden Bölüm", "Talep Eden Bolum"))
   colProje = FindHeaderCol(wsVeriSeti, headerRow, Array("Proje", "Proje Kodu", "Proje Kod"))
   colGiderAdi = FindHeaderCol(wsVeriSeti, headerRow, Array("Gider Adı", "Gider Adi", "Gider", "Hesap Adı", "Hesap Adi", "Masraf Kalemi", "Harcama Kalemi", "Kategori", "Kategori Adı", "Kategori Adi"))
   colStatu = FindHeaderCol(wsVeriSeti, headerRow, Array("Statü", "Statu", "Durum", "Talep Statüsü", "Talep Statusu", "Tedarik Statüsü", "Tedarik Statusu"))
   colGirisTarihi = FindHeaderCol(wsVeriSeti, headerRow, Array("Giriş Tarihi", "Giris Tarihi"))
   colTalepToplamUSD = FindHeaderCol(wsVeriSeti, headerRow, Array("Talep Toplam USD"))

   If colBolum = 0 Then Err.Raise vbObjectError + 1404, , "KOKPİT veri setinde 'Talep Eden Bölüm' sütunu bulunamadı."
   If colProje = 0 Then Err.Raise vbObjectError + 1405, , "KOKPİT veri setinde 'Proje' sütunu bulunamadı."
   If colStatu = 0 Then Err.Raise vbObjectError + 1406, , "KOKPİT veri setinde 'Statü' sütunu bulunamadı."
   If colGirisTarihi = 0 Then Err.Raise vbObjectError + 1407, , "KOKPİT veri setinde 'Giriş Tarihi' sütunu bulunamadı."

       statu = CStr(wsVeriSeti.Cells(i, colStatu).Value)

       If IsIsletmeOzetStatus(statu, humaStatuleriDahil) Then
           ayNo = GetMonthNoFromValue(wsVeriSeti.Cells(i, colGirisTarihi).Value)

           If ayNo >= 1 And ayNo <= 12 Then
               bolum = Trim(CStr(wsVeriSeti.Cells(i, colBolum).Value))
               projeKodu = NormalizeProjectCode(wsVeriSeti.Cells(i, colProje).Value)
               projectFinName = GetProjectFinNameFromDict(dictProjectFinName, projeKodu)
               giderAdlari = GetGiderAdlariFromProjectDict(dictProjeGiderAdi, projectFinName)

               If Len(giderAdlari) = 0 And colGiderAdi > 0 Then
                   giderAdlari = Trim(CStr(wsVeriSeti.Cells(i, colGiderAdi).Value))
               End If

               tutar = ToDoubleSafe(wsVeriSeti.Cells(i, colTalepToplamUSD).Value)

               If Len(bolum) > 0 And Len(giderAdlari) > 0 Then
                   giderParcalari = Split(giderAdlari, Chr(30))
                   For Each giderItem In giderParcalari
                       giderAdi = Trim(CStr(giderItem))
                       If Len(giderAdi) > 0 Then
                           key = NormalizeText(bolum) & "|" & NormalizeText(giderAdi) & "|" & CStr(ayNo)
                           AddToDict dict, key, tutar

                           key = NormalizeText("0V4") & "|" & NormalizeText(giderAdi) & "|" & CStr(ayNo)
                           AddToDict dict, key, tutar
                       End If
                   Next giderItem
               End If
           End If
       End If
   Next i

Private Sub SortWorksheetsAZ(ByVal wb As Workbook)

   Dim i As Long
   Dim j As Long

   If wb Is Nothing Then Exit Sub

   For i = 1 To wb.Worksheets.Count - 1
       For j = i + 1 To wb.Worksheets.Count
           If NormalizeText(wb.Worksheets(j).Name) < NormalizeText(wb.Worksheets(i).Name) Then
               wb.Worksheets(j).Move Before:=wb.Worksheets(i)
           End If
       Next j
   Next i

End Sub

Private Function CleanWorksheetName(ByVal valueIn As Variant) As String

   Dim s As String

   s = Trim(CStr(valueIn))
   s = Replace(s, ":", " ")
   s = Replace(s, "\", " ")
   s = Replace(s, "/", " ")
   s = Replace(s, "?", " ")
   s = Replace(s, "*", " ")
   s = Replace(s, "[", " ")
   s = Replace(s, "]", " ")

   Do While InStr(s, "  ") > 0
       s = Replace(s, "  ", " ")
   Loop

   If Len(s) = 0 Then s = "Bölüm"
   If Len(s) > 31 Then s = Left(s, 31)

   CleanWorksheetName = s

End Function

Private Function GetUniqueWorksheetName(ByVal wb As Workbook, ByVal baseName As String, ByVal wsCurrent As Worksheet) As String

   Dim candidate As String
   Dim counter As Long
   Dim suffix As String

   candidate = CleanWorksheetName(baseName)
   counter = 1

   Do While WorksheetNameExists(wb, candidate, wsCurrent)
       counter = counter + 1
       suffix = "_" & CStr(counter)
       candidate = Left(CleanWorksheetName(baseName), 31 - Len(suffix)) & suffix
   Loop

   GetUniqueWorksheetName = candidate

End Function

Private Function WorksheetNameExists(ByVal wb As Workbook, ByVal sheetName As String, ByVal wsCurrent As Worksheet) As Boolean

   Dim ws As Worksheet

   If wb Is Nothing Then Exit Function

   For Each ws In wb.Worksheets
       If Not ws Is wsCurrent Then
           If NormalizeText(ws.Name) = NormalizeText(sheetName) Then
               WorksheetNameExists = True
               Exit Function
           End If
       End If
   Next ws

End Function

Private Sub DeleteColumnsByExactHeaders(ByVal ws As Worksheet, ByVal headerRow As Long, ByVal headersToDelete As Variant)

   Dim lastCol As Long
   Dim c As Long
   Dim h As Variant
   Dim cellText As String
   Dim targetText As String
   Dim shouldDelete As Boolean

   lastCol = ws.Cells(headerRow, ws.Columns.Count).End(xlToLeft).Column

   For c = lastCol To 1 Step -1

       cellText = NormalizeText(ws.Cells(headerRow, c).Value)
       shouldDelete = False

       For Each h In headersToDelete
           targetText = NormalizeText(CStr(h))

           End If
       Next h

       If shouldDelete Then
           ws.Columns(c).Delete
       End If

   Next c

End Sub

Private Function IsHarcamaStatus(ByVal statu As String) As Boolean

      InStr(1, s, "PO HENUZ KESILMEDI", vbTextCompare) > 0 Then
       IsHarcamaStatus = True
       Exit Function
   End If

   If InStr(1, s, "TESLIM EDILDI", vbTextCompare) > 0 Then
       IsHarcamaStatus = True
       Exit Function
   End If

   If InStr(1, s, "TESLIMAT GECIKTI", vbTextCompare) > 0 Then
       IsHarcamaStatus = True
       Exit Function
   End If

   If InStr(1, s, "TESLIMAT ICIN VAKIT VAR", vbTextCompare) > 0 Then
       IsHarcamaStatus = True
       Exit Function
   End If

   If InStr(1, s, "KISMI TESLIMAT", vbTextCompare) > 0 Then
       IsHarcamaStatus = True
       Exit Function
   End If

   IsHarcamaStatus = False

End Function

Private Function GetMonthNoFromValue(ByVal valueIn As Variant) As Integer

   Dim s As String
   Dim parts As Variant

   On Error GoTo Hata

   If IsDate(valueIn) Then
       GetMonthNoFromValue = Month(CDate(valueIn))
       Exit Function
   End If

   s = Trim(CStr(valueIn))

   If Len(s) = 0 Then
       GetMonthNoFromValue = 0
       Exit Function
   End If

   If InStr(1, s, ".", vbTextCompare) > 0 Then

       parts = Split(s, ".")

       If UBound(parts) >= 2 Then
           GetMonthNoFromValue = CInt(parts(1))
           Exit Function
       End If

   End If

   If InStr(1, s, "-", vbTextCompare) > 0 Then

       parts = Split(s, "-")

       If UBound(parts) >= 2 Then
           GetMonthNoFromValue = CInt(parts(1))
           Exit Function
       End If

   End If

Hata:
   GetMonthNoFromValue = 0

End Function

Private Function EnsureBudgetUsageFolder(ByVal baseFolder As String, ByVal tarihMetni As String) As String

   Dim folderName As String
   Dim normalizedBaseName As String
   Dim baseName As String
   Dim p As Long

   folderName = "Bütçe Kullanım Durumu " & tarihMetni
   baseFolder = Trim(baseFolder)

   If Right(baseFolder, 1) = Application.PathSeparator Then
       baseFolder = Left(baseFolder, Len(baseFolder) - 1)
   End If

   p = InStrRev(baseFolder, Application.PathSeparator)

   If p > 0 Then
       baseName = Mid(baseFolder, p + 1)
   Else
       baseName = baseFolder
   End If

   normalizedBaseName = NormalizeText(baseName)

   If normalizedBaseName = NormalizeText(folderName) Then
       EnsureBudgetUsageFolder = baseFolder
   Else
       EnsureBudgetUsageFolder = baseFolder & Application.PathSeparator & folderName
   End If

   If Dir(EnsureBudgetUsageFolder, vbDirectory) = "" Then
       MkDir EnsureBudgetUsageFolder
   End If

End Function

Private Function BuildMudurlukProjeButceDict(ByVal wsButceKaynak As Worksheet, _
                                             ByVal dictKullanilacakButceTipleri As Object) As Object

   Dim dict As Object
   Dim headerRow As Long
   Dim lastRow As Long
   Dim colProje As Long
   Dim colDeptNo As Long
   Dim colGiderAdi As Long
   Dim colGrup As Long
   Dim colToplamUsd As Long
   Dim i As Long
   Dim projeAdi As String
   Dim deptNo As String
   Dim mudurluk As String
   Dim giderAdi As String
   Dim grup As String
   Dim lastProjeAdi As String
   Dim lastDeptNo As String
   Dim lastGiderAdi As String
   Dim lastGrup As String
   Dim kullanilacakKey As String
   Dim butceKey As String
   Dim tutar As Double

   Set dict = CreateObject("Scripting.Dictionary")

   If wsButceKaynak Is Nothing Then
       Set BuildMudurlukProjeButceDict = dict
       Exit Function
   End If

   If dictKullanilacakButceTipleri Is Nothing Then
       Set dictKullanilacakButceTipleri = CreateObject("Scripting.Dictionary")
   End If

   headerRow = FindHeaderRow(wsButceKaynak, Array("Proje", "Proje Adı", "Proje Adi", "Dept. No", "Dept No"))
   If headerRow = 0 Then
       Set BuildMudurlukProjeButceDict = dict
       Exit Function
   End If

   colProje = FindHeaderCol(wsButceKaynak, headerRow, Array("Proje", "Proje Adı", "Proje Adi"))
   colDeptNo = FindHeaderCol(wsButceKaynak, headerRow, Array("Dept. No", "Dept No", "Dept", "Departman No"))
   colGiderAdi = FindHeaderCol(wsButceKaynak, headerRow, Array("Gider Adı", "Gider Adi"))
   colGrup = FindHeaderCol(wsButceKaynak, headerRow, Array("Grup"))
   colToplamUsd = FindHeaderCol(wsButceKaynak, headerRow, Array("Toplam USD", "Toplam USD Bütçe", "Toplam USD Butce"))

   If colProje = 0 Or colDeptNo = 0 Or colGiderAdi = 0 Or colGrup = 0 Or colToplamUsd = 0 Then
       Set BuildMudurlukProjeButceDict = dict
       Exit Function
   End If

   lastRow = LastUsedRow(wsButceKaynak)

   For i = headerRow + 1 To lastRow
       projeAdi = Trim(CStr(wsButceKaynak.Cells(i, colProje).Value))
       deptNo = Trim(CStr(wsButceKaynak.Cells(i, colDeptNo).Value))
       giderAdi = Trim(CStr(wsButceKaynak.Cells(i, colGiderAdi).Value))
       grup = Trim(CStr(wsButceKaynak.Cells(i, colGrup).Value))

       If Len(projeAdi) > 0 Then lastProjeAdi = projeAdi Else projeAdi = lastProjeAdi
       If Len(deptNo) > 0 Then lastDeptNo = deptNo Else deptNo = lastDeptNo
       If Len(giderAdi) > 0 Then lastGiderAdi = giderAdi Else giderAdi = lastGiderAdi
       If Len(grup) > 0 Then lastGrup = grup Else grup = lastGrup

       mudurluk = NormalizeMudurlukKodu(deptNo)
       kullanilacakKey = NormalizeText(giderAdi) & "|" & NormalizeText(grup)

       If Len(mudurluk) > 0 And Len(projeAdi) > 0 Then
           If dictKullanilacakButceTipleri.Exists(kullanilacakKey) Then
               butceKey = mudurluk & "|" & NormalizeProjectNameKey(projeAdi)
               tutar = ToDoubleSafe(wsButceKaynak.Cells(i, colToplamUsd).Value)
               AddToDict dict, butceKey, tutar
           End If
       End If
   Next i

   Set BuildMudurlukProjeButceDict = dict

End Function

Private Function GetMudurlukProjeButceFromDict(ByVal dictMudurlukProjeButce As Object, _
                                               ByVal mudurluk As Variant, _
                                               ByVal projeAdi As Variant) As Variant

   Dim key As String

   If dictMudurlukProjeButce Is Nothing Then
       GetMudurlukProjeButceFromDict = ""
       Exit Function
   End If

   key = NormalizeMudurlukKodu(mudurluk) & "|" & NormalizeProjectNameKey(projeAdi)

   If dictMudurlukProjeButce.Exists(key) Then
       GetMudurlukProjeButceFromDict = dictMudurlukProjeButce(key)
   Else
       GetMudurlukProjeButceFromDict = ""
   End If

End Function

Private Function GetProjeButceFromDict(ByVal dictProjeButce As Object, _
                                       ByVal projeKodu As Variant, _
                                       ByVal projeAdi As Variant) As Variant

   Dim projeAdiKey As String
   Dim projeKoduKey As String
   Dim projeKoduTextKey As String
   Dim key As Variant

   If dictProjeButce Is Nothing Then
       GetProjeButceFromDict = ""
       Exit Function
   End If

   projeAdiKey = NormalizeProjectNameKey(projeAdi)
   If Len(projeAdiKey) > 0 Then
       If dictProjeButce.Exists(projeAdiKey) Then
           GetProjeButceFromDict = dictProjeButce(projeAdiKey)
           Exit Function
       End If
   End If

   projeKoduKey = NormalizeProjectCode(projeKodu)
   If Len(projeKoduKey) > 0 Then
       If dictProjeButce.Exists(projeKoduKey) Then
           GetProjeButceFromDict = dictProjeButce(projeKoduKey)
           Exit Function
       End If
   End If

   projeKoduTextKey = NormalizeText(projeKodu)
   If Len(projeKoduTextKey) > 0 Then
       If dictProjeButce.Exists(projeKoduTextKey) Then
           GetProjeButceFromDict = dictProjeButce(projeKoduTextKey)
           Exit Function
       End If
   End If

   If Len(projeAdiKey) > 0 Then
       For Each key In dictProjeButce.Keys
           If IsProjectNameKeyMatch(CStr(key), projeAdiKey) Then
               GetProjeButceFromDict = dictProjeButce(key)
               Exit Function
           End If
       Next key
   End If

   GetProjeButceFromDict = ""

End Function

Private Function NormalizeProjectNameKey(ByVal valueIn As Variant) As String

   Dim s As String
   Dim p As Long

   s = NormalizeText(valueIn)

   p = InStr(1, s, " - ", vbTextCompare)
   If p > 0 And p < Len(s) Then
       s = Trim(Mid(s, p + 3))
   End If

   NormalizeProjectNameKey = s

End Function

Private Function IsProjectNameKeyMatch(ByVal dictKey As String, ByVal projeAdiKey As String) As Boolean

   Dim normalizedDictKey As String

   normalizedDictKey = NormalizeProjectNameKey(dictKey)

   If Len(normalizedDictKey) = 0 Or Len(projeAdiKey) = 0 Then Exit Function

   If normalizedDictKey = projeAdiKey Then
       IsProjectNameKeyMatch = True
       Exit Function
   End If

   If Len(projeAdiKey) >= 8 And InStr(1, normalizedDictKey, projeAdiKey, vbTextCompare) > 0 Then
       IsProjectNameKeyMatch = True
       Exit Function
   End If

   If Len(normalizedDictKey) >= 8 And InStr(1, projeAdiKey, normalizedDictKey, vbTextCompare) > 0 Then
       IsProjectNameKeyMatch = True
       Exit Function
   End If

End Function

Private Function ResolveProjeButceKey(ByVal projeDegeri As Variant, ByVal dictProjeAdi As Object) As String

   Dim rawValue As String
   Dim normalizedCode As String
   Dim normalizedName As String
   Dim key As Variant

   rawValue = Trim(CStr(projeDegeri))
   normalizedCode = NormalizeProjectCode(rawValue)

   If Len(normalizedCode) = 0 Then
       ResolveProjeButceKey = ""
       Exit Function
   End If

   If Not dictProjeAdi Is Nothing Then
       If dictProjeAdi.Exists(normalizedCode) Then
           ResolveProjeButceKey = normalizedCode
           Exit Function
       End If

       normalizedName = NormalizeText(rawValue)

       For Each key In dictProjeAdi.Keys
           If NormalizeText(dictProjeAdi(key)) = normalizedName Then
               ResolveProjeButceKey = CStr(key)
               Exit Function
           End If
       Next key
   End If

   ResolveProjeButceKey = normalizedCode

End Function

Private Function IsGreenFont(ByVal cell As Range) As Boolean

   Dim renk As Long
   Dim r As Long
   Dim g As Long
   Dim b As Long

   If cell Is Nothing Then Exit Function

   renk = cell.Font.Color
   r = renk Mod 256
   g = (renk \ 256) Mod 256
   b = (renk \ 65536) Mod 256

   IsGreenFont = (g > r And g > b And g >= 80)

End Function

Private Function FindWorksheetByName(ByVal wb As Workbook, ByVal sheetName As String) As Worksheet

   Dim ws As Worksheet

   If wb Is Nothing Then Exit Function

   For Each ws In wb.Worksheets
       If NormalizeText(ws.Name) = NormalizeText(sheetName) Then
           Set FindWorksheetByName = ws
           Exit Function
       End If
   Next ws

   Set FindWorksheetByName = Nothing

End Function

Private Function FindHeaderRow(ByVal ws As Worksheet, ByVal possibleHeaders As Variant) As Long

   If ws Is Nothing Then
       FindHeaderRow = 0
       Exit Function
   End If

   Dim r As Long
   Dim c As Long
   Dim lastCol As Long
   Dim h As Variant
   Dim cellText As String

   For r = 1 To Application.Min(50, LastUsedRow(ws))

       lastCol = ws.Cells(r, ws.Columns.Count).End(xlToLeft).Column

       For c = 1 To lastCol

           cellText = NormalizeText(ws.Cells(r, c).Value)

           For Each h In possibleHeaders
               If cellText = NormalizeText(CStr(h)) Then
                   FindHeaderRow = r
                   Exit Function
               End If
           Next h

       Next c

   Next r

   FindHeaderRow = 0

End Function

Private Function FindHeaderCol(ByVal ws As Worksheet, ByVal headerRow As Long, ByVal possibleHeaders As Variant) As Long

   If ws Is Nothing Or headerRow <= 0 Then
       FindHeaderCol = 0
       Exit Function
   End If

   Dim c As Long
   Dim lastCol As Long
   Dim h As Variant
   Dim cellText As String
   Dim targetText As String

   lastCol = ws.Cells(headerRow, ws.Columns.Count).End(xlToLeft).Column

   For c = 1 To lastCol

       cellText = NormalizeText(ws.Cells(headerRow, c).Value)

       For Each h In possibleHeaders
           targetText = NormalizeText(CStr(h))

           If cellText = targetText Then
               FindHeaderCol = c
               Exit Function
           End If
       Next h

   Next c

   For c = 1 To lastCol

       cellText = NormalizeText(ws.Cells(headerRow, c).Value)

       For Each h In possibleHeaders
           targetText = NormalizeText(CStr(h))

           If Len(targetText) > 2 Then
               If InStr(1, cellText, targetText, vbTextCompare) > 0 Then
                   FindHeaderCol = c
                   Exit Function
               End If
           End If
       Next h

   Next c

   FindHeaderCol = 0

End Function

Private Function LastUsedRow(ByVal ws As Worksheet) As Long

   Dim f As Range

   If ws Is Nothing Then
       LastUsedRow = 1
       Exit Function
   End If

   Set f = ws.Cells.Find(What:="*", _
                         After:=ws.Range("A1"), _
                         LookIn:=xlFormulas, _
                         LookAt:=xlPart, _
                         SearchOrder:=xlByRows, _
                         SearchDirection:=xlPrevious, _
                         MatchCase:=False)

   If f Is Nothing Then
       LastUsedRow = 1
   Else
       LastUsedRow = f.Row
   End If

End Function

Private Function LastUsedCol(ByVal ws As Worksheet) As Long

   Dim f As Range

   If ws Is Nothing Then
       LastUsedCol = 1
       Exit Function
   End If

   Set f = ws.Cells.Find(What:="*", _
                         After:=ws.Range("A1"), _
                         LookIn:=xlFormulas, _
                         LookAt:=xlPart, _
                         SearchOrder:=xlByColumns, _
                         SearchDirection:=xlPrevious, _
                         MatchCase:=False)

   If f Is Nothing Then
       LastUsedCol = 1
   Else
       LastUsedCol = f.Column
   End If

End Function

End Function

Private Function NormalizeText(ByVal valueIn As Variant) As String

   Dim s As String

   If IsError(valueIn) Then
       NormalizeText = ""
       Exit Function
   End If

   s = CStr(valueIn)

   s = Replace(s, Chr(10), " ")
   s = Replace(s, Chr(13), " ")
   s = Replace(s, vbTab, " ")

   s = Trim(s)

   Do While InStr(s, "  ") > 0
       s = Replace(s, "  ", " ")
   Loop

   s = UCase$(s)

   s = Replace(s, "İ", "I")
   s = Replace(s, "İ", "I")
   s = Replace(s, "Ş", "S")
   s = Replace(s, "Ğ", "G")
   s = Replace(s, "Ü", "U")
   s = Replace(s, "Ö", "O")
   s = Replace(s, "Ç", "C")
   s = Replace(s, "ı", "I")
   s = Replace(s, "ş", "S")
   s = Replace(s, "ğ", "G")
   s = Replace(s, "ü", "U")
   s = Replace(s, "ö", "O")
   s = Replace(s, "ç", "C")

   NormalizeText = s

End Function

Private Function ToDoubleSafe(ByVal valueIn As Variant) As Double

   Dim s As String
   Dim lastComma As Long
   Dim lastDot As Long
   Dim negativeValue As Boolean

   If IsError(valueIn) Then
       ToDoubleSafe = 0
       Exit Function
   End If

   If IsNumeric(valueIn) Then
       ToDoubleSafe = CDbl(valueIn)
       Exit Function
   End If

   s = Trim(CStr(valueIn))

   If Len(s) = 0 Then
       ToDoubleSafe = 0
       Exit Function
   End If

   If Left(s, 1) = "(" And Right(s, 1) = ")" Then
       negativeValue = True
       s = Mid(s, 2, Len(s) - 2)
   End If

   s = Replace(s, "USD", "", , , vbTextCompare)
   s = Replace(s, "ABD", "", , , vbTextCompare)
   s = Replace(s, "$", "")
   s = Replace(s, " ", "")

   lastComma = InStrRev(s, ",")
   lastDot = InStrRev(s, ".")

   If lastComma > 0 And lastDot > 0 Then

       If lastComma > lastDot Then
           s = Replace(s, ".", "")
           s = Replace(s, ",", ".")
       Else
           s = Replace(s, ",", "")
       End If

   ElseIf lastComma > 0 Then

       s = Replace(s, ".", "")
       s = Replace(s, ",", ".")

   Else

       s = Replace(s, ",", "")

   End If

   If IsNumeric(s) Then
       ToDoubleSafe = CDbl(s)
       If negativeValue Then ToDoubleSafe = -ToDoubleSafe
   Else
       ToDoubleSafe = 0
   End If

End Function

Private Sub ApplyNumberFormat(ByVal ws As Worksheet, ByVal headerRow As Long, ByVal lastRow As Long, ByVal cols As Variant)

   Dim i As Long
   Dim c As Long

   For i = LBound(cols) To UBound(cols)
       c = CLng(cols(i))
       ws.Range(ws.Cells(headerRow - 1, c), ws.Cells(lastRow, c)).NumberFormat = "#,##0.00 ""USD"""
   Next i

End Sub


Private Function BuildProjeAdiDict(ByVal wsProjeBilgi As Worksheet) As Object

    Dim dict As Object
    Dim lastRow As Long
    Dim headerRow As Long
    Dim colProjeKodu As Long
    Dim colProjeAdi As Long

    Set dict = CreateObject("Scripting.Dictionary")

    lastRow = LastUsedRow(wsProjeBilgi)

    headerRow = FindHeaderRow(wsProjeBilgi, Array("Proje Kodu", "Proje Kod", "Proje"))

    If headerRow > 0 Then
        colProjeKodu = FindHeaderCol(wsProjeBilgi, headerRow, Array("Proje Kodu", "Proje Kod", "Proje"))
        colProjeAdi = FindHeaderCol(wsProjeBilgi, headerRow, Array("Proje Adı", "Proje Adi"))

        If colProjeKodu > 0 And colProjeAdi > 0 Then
            AddProjeAdiRangeToDict dict, wsProjeBilgi, headerRow + 1, lastRow, colProjeKodu, colProjeAdi
        End If
    End If

    ' Eski/standart kaynak dosya yerleşimleri için fallback:
    ' A sütunu = Proje Kodu, G sütunu = Proje Adı
    ' B sütunu = Proje Kodu, D sütunu = Proje Adı
    AddProjeAdiRangeToDict dict, wsProjeBilgi, 2, lastRow, 1, 7
    AddProjeAdiRangeToDict dict, wsProjeBilgi, 2, lastRow, 2, 4

    Set BuildProjeAdiDict = dict

End Function

Private Sub AddProjeAdiRangeToDict(ByVal dict As Object, _
                                    ByVal wsProjeBilgi As Worksheet, _
                                    ByVal firstRow As Long, _
                                    ByVal lastRow As Long, _
                                    ByVal colProjeKodu As Long, _
                                    ByVal colProjeAdi As Long)

    Dim i As Long
    Dim projeKodu As String
    Dim projeAdi As String

    If firstRow > lastRow Then Exit Sub
    If colProjeKodu <= 0 Or colProjeAdi <= 0 Then Exit Sub

    For i = firstRow To lastRow

        projeKodu = NormalizeProjectCode(wsProjeBilgi.Cells(i, colProjeKodu).Value)
        projeAdi = Trim(CStr(wsProjeBilgi.Cells(i, colProjeAdi).Value))

        If Len(projeKodu) > 0 Then
            If Not dict.Exists(projeKodu) Then
                dict.Add projeKodu, projeAdi
            ElseIf Len(Trim(CStr(dict(projeKodu)))) = 0 And Len(projeAdi) > 0 Then
                dict(projeKodu) = projeAdi
            End If
        End If

    Next i

End Sub

Private Function GetProjeAdiFromDict(ByVal dictProjeAdi As Object, ByVal projeKodu As String) As String

    Dim key As String

    key = NormalizeProjectCode(projeKodu)

    If Not dictProjeAdi Is Nothing Then
        If dictProjeAdi.Exists(key) Then
            If Len(Trim(CStr(dictProjeAdi(key)))) > 0 Then
                GetProjeAdiFromDict = CStr(dictProjeAdi(key))
                Exit Function
            End If
        End If
    End If

    GetProjeAdiFromDict = key

End Function

Private Function NormalizeProjectCode(ByVal valueIn As Variant) As String

    NormalizeProjectCode = Trim(CStr(valueIn))

End Function

Private Sub KeepOnlyKaynakSistemInVeriSeti(ByVal ws As Worksheet, ByVal hedefKaynakSistem As String)

   Dim headerRow As Long
   Dim colKaynakSistem As Long
   Dim lastRow As Long
   Dim i As Long
   Dim deger As String
   Dim hedef As String

   headerRow = 1

   colKaynakSistem = FindHeaderCol(ws, headerRow, Array("Kaynak Sistem"))

   If colKaynakSistem = 0 Then
       Err.Raise vbObjectError + 1201, , "'Kaynak Sistem' sütunu bulunamadı. Veri Seti filtreleme işlemi uygulanamadı."
   End If

   hedef = NormalizeText(hedefKaynakSistem)
   lastRow = LastUsedRow(ws)

   For i = lastRow To headerRow + 1 Step -1

       deger = NormalizeText(ws.Cells(i, colKaynakSistem).Value)

       If deger <> hedef Then
           ws.rows(i).Delete
       End If

   Next i

End Sub
